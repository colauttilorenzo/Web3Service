  3. Calcola l’hash e notarizza

    app/actions.ts
    'use server';

import { contractAddress, web3 } from '@/lib/web3';
import { notarizerAbi } from '@/lib/notarizer-abi';

export async function notarize(hash: string, metadata?: string) {
  if (!/^[0-9a-f]{64}$/.test(hash)) throw new Error('hash deve essere uno SHA-256 esadecimale.');
  return web3.getContract(notarizerAbi, contractAddress).send('mint', { recipient: contractAddress, hash, algorithm: 'sha2', metadataURI: metadata ?? '' });
}

// Server Component: la lettura avviene sul server durante il rendering.
export default async function DocumentPage({ params }: { params: Promise<{ hash: string }> }) {
  const { hash } = await params;

}

/** ABI minimo: mint per notarizzare, metadata per rileggere. Usa l'ABI completo del tuo contratto. */
export const notarizerAbi = [
  {
    type: 'function', name: 'mint', stateMutability: 'nonpayable', outputs: [],

      { name: 'recipient', type: 'address' },

  },
] as const satisfies Abi;
lib/sha256.ts
/** SHA-256 esadecimale di un file, calcolato nel browser con Web Crypto. */
export async function sha256(file: Blob): Promise<string> {
  const digest = await crypto.subtle.digest('SHA-256', await file.arrayBuffer());
  return [...new Uint8Array(digest)].map(byte => byte.toString(16).padStart(2, '0')).join('');
}
