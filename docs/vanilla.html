Integrazioni

JavaScript vanilla

Una pagina HTML, un tag script, nessun bundler.

Copia dist/browser/web3-service.min.js accanto alla pagina oppure caricalo da CDN. Il file definisce il globale CertifieWeb3.

index.html
<!doctype html>
<html lang="it">
<head>
  <meta charset="utf-8" />
  <title>Notarizza un documento</title>
  <script src="https://cdn.jsdelivr.net/npm/@certifie/web3-service@0.1.0/dist/browser/web3-service.min.js"></script>
</head>
<body>
  <input id="file" type="file" />
  <p id="status"></p>

  <script>
    const { Web3Service, Web3Error } = CertifieWeb3;
    const CONTRACT = '0x...';
    const ABI = [{
      type: 'function', name: 'mint', stateMutability: 'nonpayable', outputs: [],
      inputs: [
        { name: 'recipient', type: 'address' }, { name: 'hash', type: 'string' },
        { name: 'algorithm', type: 'string' }, { name: 'metadataURI', type: 'string' },
      ],
    }];

    const status = document.getElementById('status');
    const service = window.ethereum && new Web3Service(
      { rpcURI: 'https://rpc-amoy.polygon.technology', chainId: 80002, chainName: 'Polygon Amoy',
        nativeCurrency: { name: 'POL', symbol: 'POL', decimals: 18 } },
      { signer: { wallet: window.ethereum } },
    );

    async function sha256(file) {
      const digest = await crypto.subtle.digest('SHA-256', await file.arrayBuffer());
      return [...new Uint8Array(digest)].map(b => b.toString(16).padStart(2, '0')).join('');
    }

    document.getElementById('file').addEventListener('change', async event => {
      const file = event.target.files[0];
      if (!file) return;
      if (!service) return void (status.textContent = 'Installa un wallet come MetaMask.');
      try {
        status.textContent = 'Conferma la transazione nel wallet…';
        const result = await service.mint(ABI, CONTRACT, { hash: { data: await sha256(file), algorithm: 'sha2' }, metadata: file.name });
        status.textContent = `Notarizzato: ${result.tx.transactionHash}`;
      } catch (error) {
        status.textContent = error instanceof Web3Error && error.code === 'USER_REJECTED' ? 'Firma annullata.' : error.message;
      }
    });
  </script>
</body>
</html>

Script Node senza TypeScript

notarize.mjs
import { readFile } from 'node:fs/promises';
import { createHash } from 'node:crypto';
import { Web3Service, Web3Error } from '@certifie/web3-service';

const service = new Web3Service(process.env.RPC_URL, { signer: { privateKey: process.env.PRIVATE_KEY } });
const hash = createHash('sha256').update(await readFile(process.argv[2])).digest('hex');

try {
  const result = await service.mint(JSON.parse(await readFile('abi.json', 'utf8')), process.env.CONTRACT_ADDRESS, {
    hash: { data: hash, algorithm: 'sha2' },
  });
  console.log(JSON.stringify(result, null, 2));
} catch (error) {
  console.error(error instanceof Web3Error ? error.toJSON() : error);
  process.exitCode = 1;
}
