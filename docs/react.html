React

Notarizzazione dal browser con MetaMask, o tramite il tuo backend.

Il componente calcola l’hash del file nel browser e chiede all’utente di firmare con il suo wallet. Il file non lascia mai il dispositivo: sulla blockchain finisce solo l’hash.

  1. Installa

    npm install @certifie/web3-service
    pnpm add @certifie/web3-service
    yarn add @certifie/web3-service
    bun add @certifie/web3-service

  2. Crea il servizio

    src/use-web3-service.ts
    import { useMemo } from 'react';
import { Web3Service, type Eip1193Provider } from '@certifie/web3-service';

declare global {
  interface Window { ethereum?: Eip1193Provider }
}

export function useWeb3Service(): Web3Service | null {
  return useMemo(() => {
    if (!window.ethereum) return null; // nessun wallet installato
    return new Web3Service(
      {
        rpcURI: import.meta.env.VITE_RPC_URL,
        chainId: import.meta.env.VITE_CHAIN_ID,
        chainName: 'Polygon Amoy',
        nativeCurrency: { name: 'POL', symbol: 'POL', decimals: 18 },
        blockExplorerURI: 'https://amoy.polygonscan.com',
      },
      { signer: { wallet: window.ethereum } },
    );
  }, []);
}

  3. Notarizza un file

    src/NotarizeFile.tsx
    import { useState } from 'react';
import { Web3Error, type ITransaction } from '@certifie/web3-service';
import { useWeb3Service } from './use-web3-service';
import { notarizerAbi } from './notarizer-abi';
import { sha256 } from './sha256';

type State = { step: 'idle' | 'hashing' | 'signing' } | { step: 'done'; result: ITransaction } | { step: 'error'; message: string };

export function NotarizeFile() {
  const service = useWeb3Service();
  const [state, setState] = useState<State>({ step: 'idle' });

  async function notarize(file: File) {
    if (!service) return;
    try {
      setState({ step: 'hashing' });
      const hash = await sha256(file);
      setState({ step: 'signing' });
      const result = await service.mint(notarizerAbi, import.meta.env.VITE_CONTRACT_ADDRESS, {
        hash: { data: hash, algorithm: 'sha2' },
        metadata: file.name,
      });
      setState({ step: 'done', result });
    } catch (error) {
      const message = error instanceof Web3Error && error.code === 'USER_REJECTED'
        ? 'Firma annullata nel wallet.'
        : error instanceof Error ? error.message : String(error);
      setState({ step: 'error', message });
    }
  }

  if (!service) return <p>Installa MetaMask o un altro wallet per continuare.</p>;

  return (
    <section>
      <input type="file" disabled={state.step === 'hashing' || state.step === 'signing'}
        onChange={event => event.target.files?.[0] && notarize(event.target.files[0])} />
      {state.step === 'signing' && <p>Conferma la transazione nel wallet…</p>}
      {state.step === 'done' && (
        <p>
          Notarizzato nel blocco {state.result.tx?.blockNumber}.{' '}
          <a href={`https://amoy.polygonscan.com/tx/${state.result.tx?.transactionHash}`} target="_blank">Vedi transazione</a>
        </p>
      )}
      {state.step === 'error' && <p role="alert">{state.message}</p>}
    </section>
  );
}
src/sha256.ts
/** SHA-256 esadecimale di un file, calcolato nel browser con Web Crypto. */
export async function sha256(file: Blob): Promise<string> {
  const digest = await crypto.subtle.digest('SHA-256', await file.arrayBuffer());
  return [...new Uint8Array(digest)].map(byte => byte.toString(16).padStart(2, '0')).join('');
}

Tramite il tuo backend

Se a pagare il gas deve essere l’azienda e non l’utente, il componente invia solo l’hash a un endpoint (vedi Express, Fastify o Next.js) e il backend firma con la sua chiave.

async function notarizeOnServer(file: File): Promise<ITransaction> {
  const response = await fetch('/api/notarize', {
    method: 'POST',
    headers: { 'content-type': 'application/json' },
    body: JSON.stringify({ hash: await sha256(file), metadata: file.name }),
  });
  if (!response.ok) throw new Error((await response.json()).message);
  return response.json();
}
