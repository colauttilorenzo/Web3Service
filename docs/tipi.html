API

Tipi

Le dichiarazioni TypeScript esportate dal pacchetto e le classi .NET.

Tutti i tipi si importano dall’entry point: import type { ITransaction } from '@certifie/web3-service'. La scheda TypeScript mostra la dichiarazione, la scheda C# la classe corrispondente del pacchetto NuGet, serializzata con gli stessi nomi JSON.

IEvmProvider

Models.ts
interface IEvmProvider {
  rpcURI: string;                 // http(s):// o ws(s)://
  rpcPublicURI?: string;          // fallback
  chainId?: ChainId;              // default '0x1'
  chainName?: string;             // default 'Ethereum'
  nativeCurrency?: Partial<NativeCurrency>;
  blockExplorerURI?: string;
}

interface NativeCurrency {
  name?: string;      // default 'Ether'
  symbol: string;     // default 'ETH'
  decimals: number;   // default 18, solo informativo
}

type ChainId = string | number | bigint;
type EthereumProviderConfig = IEvmProvider; // alias di compatibilità
Models.cs
public sealed class EvmProvider
{
    public string RpcURI { get; set; } = "";       // http(s):// o ws(s)://
    public string? RpcPublicURI { get; set; }      // fallback
    public string? ChainId { get; set; }           // "1337" o "0x539", default "0x1"
    public string? ChainName { get; set; }         // default "Ethereum"
    public NativeCurrency? NativeCurrency { get; set; }
    public string? BlockExplorerURI { get; set; }
}

public sealed class NativeCurrency
{
    public string? Name { get; set; }   // default "Ether"
    public string? Symbol { get; set; } // default "ETH"
    public int? Decimals { get; set; }  // default 18, solo informativo
}

Web3ServiceOptions

Models.ts
interface Web3ServiceOptions {
  signer?: SignerOptions;
  timeoutMs?: number;            // 30000
  transactionTimeoutMs?: number; // 120000
  pollingIntervalMs?: number;    // 1000
  gasBuffer?: ChainId;           // 10000
  web3Factory?: (rpcURI: string) => Web3;
}

type SignerOptions =
  | { privateKey: string }
  | { mnemonic: string | { phrase: string; password?: string }; walletIndex?: number }
  | { wallet: Eip1193Provider; switchChain?: boolean };

interface Eip1193Provider {
  request(args: { method: string; params?: readonly unknown[] | object }): Promise<unknown>;
}
Models.cs
public sealed class Web3ServiceOptions
{
    public string? PrivateKey { get; set; }              // oppure Mnemonic
    public string? Mnemonic { get; set; }
    public string MnemonicPassword { get; set; } = "";
    public int WalletIndex { get; set; } = 1;            // m/44'/60'/0'/0/{WalletIndex}
    public TimeSpan Timeout { get; set; }                // 30 secondi
    public TimeSpan TransactionTimeout { get; set; }     // 120 secondi
    public TimeSpan PollingInterval { get; set; }        // 1 secondo
    public BigInteger GasBuffer { get; set; } = 10_000;
}

ITransaction

Models.ts
interface ITransaction {
  name?: 'mint' | 'deploy' | string;
  requestedAt?: string;   // millisecondi
  completedAt?: string;   // timestamp del blocco, se riuscita
  rejectedAt?: string;    // timestamp del blocco, se revertita
  tx?: ITransactionReceipt;
  error?: Web3ErrorJSON;
  nativeFee?: { amount: string; symbol: string; decimals: string };
}
Models.cs
public sealed class TransactionResult
{
    public string? Name { get; set; }           // "mint", "deploy"
    public string? RequestedAt { get; set; }    // millisecondi
    public string? CompletedAt { get; set; }    // timestamp del blocco, se riuscita
    public string? RejectedAt { get; set; }     // timestamp del blocco, se revertita
    public TransactionReceipt? Tx { get; set; }
    public Web3ErrorInfo? Error { get; set; }
    public NativeFee? NativeFee { get; set; }   // Amount, Symbol, Decimals
}

ITransactionReceipt

Models.ts
interface ITransactionReceipt {
  transactionHash: string;
  blockHash: string;
  blockNumber: string;
  transactionIndex: string;
  from: string;
  to?: string | null;
  contractAddress?: string | null;
  type?: string;
  status: string;            // '1' riuscita, '0' revertita
  cumulativeGasUsed: string;
  gasUsed: string;
  effectiveGasPrice?: string;
  logsBloom: string;
  logs: ITransactionLog[];
}

interface ITransactionLog {
  address?: string; data?: string; topics?: string[];
  logIndex?: string; transactionIndex?: string; transactionHash?: string;
  blockHash?: string; blockNumber?: string; removed?: boolean;
}
Models.cs
public sealed class TransactionReceipt
{
    public string TransactionHash { get; set; } = "";
    public string BlockHash { get; set; } = "";
    public string BlockNumber { get; set; } = "";
    public string TransactionIndex { get; set; } = "";
    public string From { get; set; } = "";
    public string? To { get; set; }
    public string? ContractAddress { get; set; }
    public string? Type { get; set; }
    public string Status { get; set; } = "";    // "1" riuscita, "0" revertita
    public string CumulativeGasUsed { get; set; } = "";
    public string GasUsed { get; set; } = "";
    public string? EffectiveGasPrice { get; set; }
    public string LogsBloom { get; set; } = "";
    public List<TransactionLog> Logs { get; set; } = new List<TransactionLog>();
}
