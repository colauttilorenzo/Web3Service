API

Tipi

Le dichiarazioni TypeScript esportate dal pacchetto e le classi .NET.

Tutti i tipi si importano dall’entry point: import type { ITransaction } from '@certifie/web3-service'. La scheda TypeScript mostra la dichiarazione, la scheda C# la classe corrispondente del pacchetto NuGet, serializzata con gli stessi nomi JSON.

IEvmProvider

Models.ts
interface IEvmProvider {
  kind?: 'evm';                   // facoltativo: senza kind il provider è EVM
  rpcURI: string;                 // http(s):// o ws(s)://
  rpcPublicURI?: string;          // fallback
  chainId?: ChainId;              // default '0x1'
  chainName?: string;             // default 'Ethereum'
  nativeCurrency?: Partial<NativeCurrency>;
  blockExplorerURI?: string;
}

interface NativeCurrency {
  name?: string;      // default 'Ether'
  symbol: string;     // default 'ETH'
  decimals: number;   // default 18, solo informativo
}

type ChainId = string | number | bigint;
type EthereumProviderConfig = IEvmProvider; // alias di compatibilità
Models.cs
public sealed class EvmProvider : IEvmProvider
{
    public string RpcURI { get; set; } = "";       // http(s):// o ws(s)://
    public string? RpcPublicURI { get; set; }      // fallback
    public string? ChainId { get; set; }           // "1337" o "0x539", default "0x1"
    public string? ChainName { get; set; }         // default "Ethereum"
    public NativeCurrency? NativeCurrency { get; set; }
    public string? BlockExplorerURI { get; set; }
}

public sealed class NativeCurrency
{
    public string? Name { get; set; }   // default "Ether"
    public string? Symbol { get; set; } // default "ETH"
    public int? Decimals { get; set; }  // default 18, solo informativo
}

IAvmProvider e IBitcoinProvider

Il tipo del provider sceglie la blockchain: vedi Più blockchain. In TypeScript lo indica il campo kind, in .NET l’interfaccia implementata.

Models.ts
interface IAvmProvider {
  kind: 'avm';
  rpcURI: string;          // algod
  rpcPublicURI?: string;   // fallback
  token?: string;          // X-Algo-API-Token
  indexerURI?: string;
  indexerToken?: string;
  chainId?: string;        // genesis ID, default 'mainnet-v1.0'
  chainName?: string;      // default 'Algorand'
  nativeCurrency?: Partial<NativeCurrency>; // default Algo, ALGO, 6
  blockExplorerURI?: string;
}

interface IBitcoinProvider {
  kind: 'bitcoin';
  rpcURI: string;          // API Esplora
  rpcPublicURI?: string;   // fallback
  chainId?: string;        // 'mainnet' (default), 'testnet', 'testnet4', 'signet', 'regtest'
  chainName?: string;      // default 'Bitcoin'
  nativeCurrency?: Partial<NativeCurrency>; // default Bitcoin, BTC, 8
  blockExplorerURI?: string;
}

type ProviderInput = IEvmProvider | IAvmProvider | IBitcoinProvider | string; // una stringa è l'URI di un nodo EVM
type ChainKind = 'evm' | 'avm' | 'bitcoin';
Models.cs
public interface IBlockchainProvider // campi comuni; l'interfaccia derivata sceglie la blockchain
{
    string RpcURI { get; }
    string? RpcPublicURI { get; }
    string? ChainId { get; }
    string? ChainName { get; }
    NativeCurrency? NativeCurrency { get; }

}

public interface IEvmProvider : IBlockchainProvider { }
public interface IBitcoinProvider : IBlockchainProvider { }
public interface IAvmProvider : IBlockchainProvider
{
    string? Token { get; }         // X-Algo-API-Token
    string? IndexerURI { get; }
    string? IndexerToken { get; }
}

// Implementazioni pronte, con le stesse proprietà in lettura e scrittura
public sealed class EvmProvider : IEvmProvider { /* … */ }
public sealed class AvmProvider : IAvmProvider { /* … */ }         // ChainId default "mainnet-v1.0"
public sealed class BitcoinProvider : IBitcoinProvider { /* … */ } // ChainId default "mainnet"

Web3ServiceOptions

Models.ts
interface Web3ServiceOptions {
  signer?: SignerOptions;
  timeoutMs?: number;            // 30000
  transactionTimeoutMs?: number; // 120000
  pollingIntervalMs?: number;    // 1000
  gasBuffer?: ChainId;           // 10000
  web3Factory?: (rpcURI: string) => Web3;
}

type SignerOptions =
  | { privateKey: string }                       // su Algorand: seed 0x… o secret key base64
  | { mnemonic: string | { phrase: string; password?: string }; walletIndex?: number } // su Algorand: 25 parole
  | { wallet: Eip1193Provider | IAvmWallet; switchChain?: boolean };

interface Eip1193Provider {
  request(args: { method: string; params?: readonly unknown[] | object }): Promise<unknown>;
}

interface IAvmWallet { // wallet Algorand, es. da @txnlab/use-wallet
  address: string;
  signTransactions(transactions: Uint8Array[]): Promise<(Uint8Array | null)[]>;
}
Models.cs
public sealed class Web3ServiceOptions
{
    public string? PrivateKey { get; set; }              // oppure Mnemonic; su Algorand seed 0x… o secret key base64
    public string? Mnemonic { get; set; }                // su Algorand: le 25 parole
    public string MnemonicPassword { get; set; } = "";
    public int WalletIndex { get; set; } = 1;            // m/44'/60'/0'/0/{WalletIndex}
    public TimeSpan Timeout { get; set; }                // 30 secondi
    public TimeSpan TransactionTimeout { get; set; }     // 120 secondi
    public TimeSpan PollingInterval { get; set; }        // 1 secondo
    public BigInteger GasBuffer { get; set; } = 10_000;
}

IContractMethod

Models.ts
type MethodKind = 'read' | 'write' | 'payable';

interface IContractMethod {
  name: string;
  signature: string;                  // 'transfer(address,uint256)'
  kind: MethodKind;                   // read: gratuita; write: transazione; payable: transazione che riceve moneta
  stateMutability: 'pure' | 'view' | 'nonpayable' | 'payable';
  parameters: Record<string, string>; // nome → tipo JSON: string, number, boolean, array, object
  inputs: Record<string, string>;     // nome → tipo Solidity: address, uint256, ...
  outputs: Record<string, string>;    // nome → tipo Solidity; senza nome: result0, result1, ...
}

// Parametri per nome: interi come number o stringa decimale, indirizzi e bytes come stringhe 0x.
type ContractParameters = Record<string, unknown>;
Models.cs
public sealed class ContractMethod
{
    public string Name { get; set; }
    public string Signature { get; set; }                      // "transfer(address,uint256)"
    public string Kind { get; set; }                           // "read", "write" o "payable"
    public string StateMutability { get; set; }                // "pure", "view", "nonpayable", "payable"
    public Dictionary<string, string> Parameters { get; set; } // nome → tipo JSON
    public Dictionary<string, string> Inputs { get; set; }     // nome → tipo Solidity
    public Dictionary<string, string> Outputs { get; set; }
}
