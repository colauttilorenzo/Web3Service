Next.js

Route Handler con firma lato server e componente client.

La chiave resta sul server: il Route Handler firma, il componente client invia solo l’hash. L’istanza è creata a livello di modulo, così le scritture dello stesso processo ricevono nonce distinti.

  1. Istanza condivisa

    lib/web3.ts
    import 'server-only';
import { Web3Service } from '@certifie/web3-service';

export const web3 = new Web3Service(
  {
    rpcURI: process.env.RPC_URL!,
    rpcPublicURI: process.env.RPC_FALLBACK_URL,
    chainId: process.env.CHAIN_ID,
  },
  { signer: { privateKey: process.env.PRIVATE_KEY! } },
);

  2. Route Handler di notarizzazione

    app/api/notarize/route.ts
    import { NextResponse } from 'next/server';
import { Web3Error } from '@certifie/web3-service';
import { web3 } from '@/lib/web3';
import { notarizerAbi } from '@/lib/notarizer-abi';
import { httpStatus } from '@/lib/http-status';

export const runtime = 'nodejs';

export async function POST(request: Request) {
  const { hash, metadata } = await request.json();
  if (typeof hash !== 'string' || !/^[0-9a-f]{64}$/.test(hash)) {
    return NextResponse.json({ message: 'hash deve essere uno SHA-256 esadecimale.' }, { status: 400 });
  }
  try {
    const result = await web3.getContract(notarizerAbi, process.env.CONTRACT_ADDRESS!).send('mint', {
      recipient: process.env.CONTRACT_ADDRESS!, hash, algorithm: 'sha2', metadataURI: metadata ?? '',
    });
    return NextResponse.json(result);
  } catch (error) {
    if (error instanceof Web3Error) return NextResponse.json(error.toJSON(), { status: httpStatus(error) });
    throw error;
  }
}

  3. Lettura della ricevuta

    app/api/transactions/[hash]/route.ts
    import { NextResponse } from 'next/server';
import { Web3Error } from '@certifie/web3-service';
import { web3 } from '@/lib/web3';
import { httpStatus } from '@/lib/http-status';

export const runtime = 'nodejs';

export async function GET(_request: Request, { params }: { params: Promise<{ hash: string }> }) {
  const { hash } = await params;
  try {
    return NextResponse.json(await web3.getTransactionReceipt(hash));
  } catch (error) {
    if (error instanceof Web3Error) return NextResponse.json(error.toJSON(), { status: httpStatus(error) });
    throw error;
  }
}

  4. Componente client

    app/notarize/notarize-form.tsx
    'use client';

import { useState } from 'react';
import type { ITransaction } from '@certifie/web3-service';
import { sha256 } from '@/lib/sha256';

export function NotarizeForm() {
  const [result, setResult] = useState<ITransaction>();
  const [error, setError] = useState<string>();

  async function onChange(file: File) {
    setError(undefined);
    const response = await fetch('/api/notarize', {
      method: 'POST',
      headers: { 'content-type': 'application/json' },
      body: JSON.stringify({ hash: await sha256(file), metadata: file.name }),
    });
    const body = await response.json();
    if (response.ok) setResult(body); else setError(body.message);
  }

  return (
    <div>
      <input type="file" onChange={event => event.target.files?.[0] && onChange(event.target.files[0])} />
      {result && <p>Transazione {result.tx?.transactionHash}, gas {result.nativeFee?.amount} wei</p>}
      {error && <p role="alert">{error}</p>}
    </div>
  );
}
Nota

Il runtime Edge non è supportato per le scritture: usa runtime = 'nodejs'. Per firmare con il wallet dell’utente usa nel componente client il codice della pagina React.
