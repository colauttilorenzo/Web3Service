Integrazioni

Angular

Servizio iniettabile con signals e componente standalone.

  1. Installa

    npm install @certifie/web3-service
    pnpm add @certifie/web3-service
    yarn add @certifie/web3-service
    bun add @certifie/web3-service

  2. Servizio

    src/app/notarization.service.ts
    import { Injectable, signal } from '@angular/core';
import { Web3Error, Web3Service, type Eip1193Provider, type ITransaction } from '@certifie/web3-service';
import { environment } from '../environments/environment';
import { notarizerAbi } from './notarizer-abi';
import { sha256 } from './sha256';

@Injectable({ providedIn: 'root' })
export class NotarizationService {
  private readonly wallet = (window as { ethereum?: Eip1193Provider }).ethereum;
  private readonly web3 = this.wallet
    ? new Web3Service(environment.evm, { signer: { wallet: this.wallet } })
    : null;

  readonly busy = signal(false);
  readonly last = signal<ITransaction | null>(null);
  readonly error = signal<string | null>(null);
  readonly hasWallet = this.web3 !== null;

  async notarize(file: File): Promise<void> {
    if (!this.web3) return;
    this.busy.set(true);
    this.error.set(null);
    try {
      const result = await this.web3.mint(notarizerAbi, environment.contractAddress, {
        hash: { data: await sha256(file), algorithm: 'sha2' },
        metadata: file.name,
      });
      this.last.set(result);
    } catch (error) {
      this.error.set(error instanceof Web3Error && error.code === 'USER_REJECTED' ? 'Firma annullata.' : String(error));
    } finally {
      this.busy.set(false);
    }
  }

  verify(hash: string) {
    return this.web3!.callContractMethod<string[]>('metadata', notarizerAbi, environment.contractAddress, [hash]);
  }
}

  3. Componente

    src/app/notarize.component.ts
    import { Component, inject } from '@angular/core';
import { NotarizationService } from './notarization.service';

@Component({
  selector: 'app-notarize',
  standalone: true,
  template: `
    @if (notarization.hasWallet) {
      <input type="file" [disabled]="notarization.busy()" (change)="onFile($event)" />
      @if (notarization.busy()) { <p>Conferma la transazione nel wallet…</p> }
      @if (notarization.last(); as result) {
        <p>Blocco {{ result.tx?.blockNumber }} · {{ result.tx?.transactionHash }}</p>
      }
      @if (notarization.error(); as message) { <p role="alert">{{ message }}</p> }
    } @else {
      <p>Installa un wallet per continuare.</p>
    }
  `,
})
export class NotarizeComponent {
  protected readonly notarization = inject(NotarizationService);

  onFile(event: Event) {
    const file = (event.target as HTMLInputElement).files?.[0];
    if (file) void this.notarization.notarize(file);
  }
}
