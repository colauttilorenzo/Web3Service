Integrazioni

.NET

Pacchetto NuGet nativo, console e ASP.NET Core.

Certifie.Web3Service è un client C# basato su Nethereum, per net8.0 e netstandard2.0. Ha gli stessi default, gli stessi codici di errore e produce lo stesso JSON della versione JavaScript.

dotnet add package Certifie.Web3Service
dotnet add package Certifie.Web3Service --source ~/Desktop/workspace/web3-service/artifacts/nuget
<PackageReference Include="Certifie.Web3Service" Version="0.1.0" />

Console

Program.cs
using System.Security.Cryptography;
using System.Text.Json;
using Certifie.Web3;

var web3 = new Web3Service(new EvmProvider
{
    RpcURI = "https://rpc-amoy.polygon.technology",
    ChainId = "80002",
    ChainName = "Polygon Amoy",
    NativeCurrency = new NativeCurrency { Name = "POL", Symbol = "POL", Decimals = 18 },
}, new Web3ServiceOptions { PrivateKey = Environment.GetEnvironmentVariable("PRIVATE_KEY") });

var abi = await File.ReadAllTextAsync("notarizer-abi.json");
var hash = Convert.ToHexString(SHA256.HashData(await File.ReadAllBytesAsync(args[0]))).ToLowerInvariant();

try
{
    var result = await web3.MintAsync(abi, Environment.GetEnvironmentVariable("CONTRACT_ADDRESS")!, new NotarizationDocument
    {
        Hash = new NotarizationHash { Data = hash, Algorithm = "sha2" },
        Metadata = Path.GetFileName(args[0]),
    });
    Console.WriteLine(JsonSerializer.Serialize(result)); // stesso JSON della libreria JS
}
catch (Web3Exception error)
{
    Console.Error.WriteLine($"{error.Code}: {error.Message} {error.TransactionHash}");
}

ASP.NET Core

Program.cs
using Certifie.Web3;

var builder = WebApplication.CreateBuilder(args);
var abi = await File.ReadAllTextAsync("notarizer-abi.json");
var contract = builder.Configuration["Web3:Contract"]!;

// Singleton: le scritture della stessa istanza sono serializzate e i nonce restano distinti.
builder.Services.AddSingleton(_ => new Web3Service(
    new EvmProvider { RpcURI = builder.Configuration["Web3:RpcURI"]!, ChainId = builder.Configuration["Web3:ChainId"] },
    new Web3ServiceOptions { PrivateKey = builder.Configuration["Web3:PrivateKey"] }));

var app = builder.Build();

app.MapPost("/notarize", async (Web3Service web3, NotarizationDocument document, CancellationToken token) =>
    Results.Ok(await web3.MintAsync(abi, contract, document, cancellationToken: token)));

app.MapGet("/transactions/{hash}", async (Web3Service web3, string hash) =>
    Results.Ok(await web3.GetTransactionReceiptAsync(hash)));

app.Use(async (context, next) =>
{
    try { await next(); }
    catch (Web3Exception error)
    {
        context.Response.StatusCode = error.Code switch
        {
            "INVALID_ARGUMENT" or "METHOD_NOT_FOUND" => 400,
            "TRANSACTION_NOT_FOUND" => 404,
            "TRANSACTION_REVERTED" => 409,
            "PROVIDER_UNAVAILABLE" or "CHAIN_ID_MISMATCH" => 503,
            _ => 502,
        };
        await context.Response.WriteAsJsonAsync(error.ToInfo());
    }
});

app.Run();
