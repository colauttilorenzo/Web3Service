Fastify

API di notarizzazione con validazione dello schema.

npm install fastify @certifie/web3-service
pnpm add fastify @certifie/web3-service
yarn add fastify @certifie/web3-service
bun add fastify @certifie/web3-service
server.ts
import Fastify from 'fastify';
import { Web3Error, Web3Service } from '@certifie/web3-service';
import { notarizerAbi } from './notarizer-abi.js';
import { httpStatus } from './http-status.js';

const web3 = new Web3Service(
  { rpcURI: process.env.RPC_URL!, rpcPublicURI: process.env.RPC_FALLBACK_URL, chainId: process.env.CHAIN_ID },
  { signer: { privateKey: process.env.PRIVATE_KEY! } },
);
const contract = process.env.CONTRACT_ADDRESS!;

const app = Fastify({ logger: true });

app.post<{ Body: { hash: string; metadata?: string } }>('/notarize', {
  schema: {
    body: {
      type: 'object',
      required: ['hash'],
      properties: { hash: { type: 'string', pattern: '^[0-9a-f]{64}$' }, metadata: { type: 'string' } },
    },
  },
}, async request => web3.mint(notarizerAbi, contract, {
  hash: { data: request.body.hash, algorithm: 'sha2' },
  metadata: request.body.metadata,
}));

app.get<{ Params: { hash: string } }>('/transactions/:hash', async request =>
  web3.getTransactionReceipt(request.params.hash));

app.get<{ Params: { hash: string } }>('/documents/:hash', async request => {
  const [hash, algorithm, metadata] = await web3.callContractMethod<string[]>('metadata', notarizerAbi, contract, [request.params.hash]);
  return { notarized: hash !== '', hash, algorithm, metadata };
});

app.get('/health', async () => web3.check());

app.setErrorHandler((error, _request, reply) => {
  if (error instanceof Web3Error) return reply.status(httpStatus(error)).send(error.toJSON());
  return reply.send(error);
});

await app.listen({ port: 3000 });
